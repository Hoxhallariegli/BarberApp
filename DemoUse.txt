### RREGULLAT PËR RELACIONET (DOT NOTATION) ###
# Kur dëshiron të marrësh një fushë (p.sh 'name') nga një tabelë e lidhur (p.sh 'employees')
# përmes një tabele ndërmjetëse (p.sh 'mechanics'), përdor rregullat e mëposhtme:
#
# 1. FORMATI: "relation_id`n8`nrelated_table`nRELATION.FIELD`nno"
# 2. SHEMBULL (JobCard -> Mechanic -> Employee):
#    Në vend të thjesht 'name', përdor 'employee.name'.
#    Gjeneratori do të bëjë automatikisht:
#    - Eager Loading: JobCard::with('mechanic.employee')
#    - View Index: $item->mechanic?->employee?->name
#    - Dropdown: Do të shfaqë emrin e punonjësit për çdo mekanik.
#
# 3. AUTO-LOAD (Cascading):
#    Nëse në një formë ke dy dropdown-e të lidhura (p.sh Model dhe Brand),
#    vendos Modelin pas Brand-it. Kur zgjedh Modelin, sistemi do të
#    mbushë automatikisht Brand-in e duhur.

use App\Models\{Customer, VehicleBrand, VehicleModel, Vehicle, VehicleDocument, Employee, Mechanic, Service, Part, JobCard, JobCardService, JobCardPart, Inventory, Supplier, PurchaseOrder, PurchaseOrderItem, Estimate, EstimateItem, Invoice, InvoiceItem, Appointment, Report};
use Illuminate\Support\Str;

echo "🚀 Duke nisur popullimin e plotë të sistemit Enterprise...\n";

// 1. Struktura e Makinave (Brands & Models)
$brands = collect(['Toyota', 'Mercedes-Benz', 'BMW', 'Audi', 'Volkswagen', 'Ford'])->map(fn($n) => VehicleBrand::create(['name' => $n]));
foreach ($brands as $brand) {
    collect(['Sedan', 'SUV', 'Coupe'])->each(fn($m) => VehicleModel::create(['name' => $brand->name . " " . $m, 'brand_id' => $brand->id]));
}
echo "✅ Markat dhe Modelet u krijuan.\n";

// 2. Njerëzit (Klientë, Furnitorë, Staf)
for ($i = 0; $i < 15; $i++) {
    Customer::create(['name' => fake()->name(), 'email' => fake()->unique()->safeEmail(), 'phone' => fake()->phoneNumber()]);
    Supplier::create(['name' => fake()->company(), 'email' => fake()->unique()->safeEmail(), 'phone' => fake()->phoneNumber()]);
    $emp = Employee::create(['name' => fake()->name(), 'email' => fake()->unique()->safeEmail(), 'phone' => fake()->phoneNumber()]);
    if ($i < 5) Mechanic::create(['employee_id' => $emp->id, 'specialization' => fake()->randomElement(['Motorist', 'Elektriçist', 'Gomonier'])]);
}
echo "✅ Klientët, Furnitorët dhe Mekanikët u krijuan.\n";

// 3. Inventari (Pjesë dhe Shërbime)
$services = collect([
    ['name' => 'Ndërrim Vaji Full', 'price' => 80.00, 'duration' => 45],
    ['name' => 'Diagnostikë Kompjuteri', 'price' => 30.00, 'duration' => 20],
    ['name' => 'Riparim Frenash', 'price' => 50.00, 'duration' => 60],
    ['name' => 'Klimatizim Gas', 'price' => 40.00, 'duration' => 30]
])->map(fn($s) => Service::create($s));

$parts = collect([
    ['name' => 'Vaj Castrol 5W30', 'price' => 15.00, 'stock' => 100],
    ['name' => 'Filtër Vaji', 'price' => 10.00, 'stock' => 50],
    ['name' => 'Pllaka Frenash', 'price' => 45.00, 'stock' => 20],
    ['name' => 'Goma Continentals', 'price' => 120.00, 'stock' => 16]
])->map(fn($p) => Part::create($p));

foreach ($parts as $p) Inventory::create(['part_id' => $p->id, 'quantity' => $p->stock]);
echo "✅ Shërbimet, Pjesët dhe Inventari u regjistruan.\n";

// 4. Fluksi i Punës (Makinat -> JobCards -> Invoices)
$customers = Customer::all();
foreach ($customers->take(10) as $customer) {
    $model = VehicleModel::inRandomOrder()->first();
    $vehicle = Vehicle::create([
        'brand_id' => $model->brand_id, 'model_id' => $model->id, 'year' => rand(2015, 2024),
        'customer_id' => $customer->id, 'license_plate' => 'AA '.rand(100,999).' '.strtoupper(Str::random(2)),
        'vin' => strtoupper(Str::random(17))
    ]);

    // Shto Dokument
    VehicleDocument::create(['type' => 'Leje Qarkullimi', 'document' => 'uploads/docs/licence_'.Str::random(5).'.pdf', 'vehicle_id' => $vehicle->id]);

    // Takim
    Appointment::create(['vehicle_id' => $vehicle->id, 'appointment_date' => now()->addDays(rand(1, 10)), 'status' => 'Pending']);

    // Kartela e Punës
    $jc = JobCard::create([
        'vehicle_id' => $vehicle->id, 'customer_id' => $customer->id,
        'mechanic_id' => Mechanic::inRandomOrder()->first()->id,
        'status' => 'Completed', 'opened_at' => now()->subHours(5), 'closed_at' => now()
    ]);

    // Lidhja me Shërbime dhe Pjesë
    $serv = $services->random();
    $prt = $parts->random();
    JobCardService::create(['job_card_id' => $jc->id, 'service_id' => $serv->id, 'quantity' => 1, 'price' => $serv->price]);
    JobCardPart::create(['job_card_id' => $jc->id, 'part_id' => $prt->id, 'quantity' => 1, 'price' => $prt->price]);

    // Estimimi
    $est = Estimate::create(['job_card_id' => $jc->id, 'estimate_date' => now()->subHours(4), 'status' => 'Approved']);
    EstimateItem::create(['estimate_id' => $est->id, 'service_id' => $serv->id, 'part_id' => $prt->id, 'quantity' => 1]);

    // Fatura Finale
    $inv = Invoice::create(['job_card_id' => $jc->id, 'invoice_date' => now(), 'total' => ($serv->price + $prt->price), 'status' => 'Paid']);
    InvoiceItem::create(['invoice_id' => $inv->id, 'service_id' => $serv->id, 'part_id' => $prt->id, 'quantity' => 1, 'price' => ($serv->price + $prt->price)]);
}
echo "✅ Procesi i punës (JobCards & Invoices) u gjenerua.\n";

// 5. Furnizimet dhe Raportet
$supplier = Supplier::first();
$po = PurchaseOrder::create(['supplier_id' => $supplier->id, 'order_date' => now(), 'status' => 'Received']);
PurchaseOrderItem::create(['purchase_order_id' => $po->id, 'part_id' => Part::first()->id, 'quantity' => 50, 'price' => 12.50]);

Report::create(['report_type' => 'Monthly Revenue', 'report_date' => now()]);
Report::create(['report_type' => 'Inventory Status', 'report_date' => now()]);

echo "\n✨ SISTEMI U MBUSH PLOTËSISHT! ✨\n";
echo "Tani mund të shkosh te Dashboard dhe të shohësh statistikat reale.\n";
